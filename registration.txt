class RegisterRequest {
  String name;
  String phone;
  String password;
  String role;

  RegisterRequest({
    required this.name,
    required this.phone,
    required this.password,
    required this.role,
  });

  Map<String, dynamic> toJson() => {
    "name": name,
    "phone": phone,
    "password": password,
    "role": role,
  };
}

🟢 STEP 6 — API Service (Laravel Backend Ready)
class AuthService {
  final Dio dio = Dio();

  Future register(RegisterRequest data) async {
    return await dio.post('/register', data: data.toJson());
  }

  Future verifyOtp(String phone, String otp) async {
    return await dio.post('/verify-otp', data: {
      "phone": phone,
      "otp": otp
    });
  }
}
🟢 STEP 7 — State Management (GetX)
class AuthController extends GetxController {
  var isLoading = false.obs;

  void registerUser(RegisterRequest data) async {
    try {
      isLoading(true);
      await AuthService().register(data);
